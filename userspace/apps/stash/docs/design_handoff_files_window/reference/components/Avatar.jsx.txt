import React from 'react';

/**
 * Avatar — user/contact avatar for AIVA OS.
 * Shows an image, or initials on a glass backing. Optional status dot.
 */
export function Avatar({
  src,
  alt = '',
  initials,
  size = 40,
  status,
  square = false,
  style = {},
}) {
  const statusColor = {
    online: 'var(--color-success)',
    busy: 'var(--color-destructive)',
    away: 'var(--color-warning)',
    offline: 'var(--glass-text-secondary)',
  }[status];

  return (
    <div style={{ position: 'relative', width: size, height: size, flexShrink: 0, ...style }}>
      <div style={{
        width: '100%', height: '100%', overflow: 'hidden',
        borderRadius: square ? Math.round(size * 0.28) : '50%',
        background: 'var(--glass-card-bg)',
        border: '1px solid var(--glass-card-border)',
        backdropFilter: 'blur(20px)', WebkitBackdropFilter: 'blur(20px)',
        display: 'flex', alignItems: 'center', justifyContent: 'center',
        boxShadow: 'inset 0 1px 1px rgba(255,255,255,0.2)',
      }}>
        {src ? (
          <img src={src} alt={alt} style={{ width: '100%', height: '100%', objectFit: 'cover', display: 'block' }} />
        ) : (
          <span style={{
            fontFamily: 'var(--font-sans)', fontWeight: 600,
            fontSize: size * 0.4, color: 'var(--glass-text-strong)', letterSpacing: '0.01em',
          }}>{initials}</span>
        )}
      </div>
      {statusColor && (
        <span style={{
          position: 'absolute', right: 0, bottom: 0,
          width: Math.max(8, size * 0.26), height: Math.max(8, size * 0.26),
          borderRadius: '50%', background: statusColor,
          border: '2px solid var(--glass-panel-bg)', boxSizing: 'border-box',
        }} />
      )}
    </div>
  );
}
