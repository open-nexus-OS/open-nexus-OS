import React from 'react';
import { WindowButton } from './WindowButton.jsx';
import { Icon } from './Icon.jsx';

/**
 * WindowControls — the minimise / maximise / close cluster that sits
 * at the right edge of every AIVA OS window title bar.
 */
export function WindowControls({ onMinimize, onMaximize, onClose, style = {} }) {
  return (
    <div style={{ display: 'flex', alignItems: 'center', gap: 1, flexShrink: 0, ...style }}>
      <WindowButton title="Minimieren" size={34} radius={10} onClick={onMinimize}>
        <Icon paths='<path d="M5 12h14"/>' size={14} strokeWidth={2.5} />
      </WindowButton>
      <WindowButton title="Maximieren" size={34} radius={10} onClick={onMaximize}>
        <Icon paths='<rect width="18" height="18" x="3" y="3" rx="2"/>' size={13} strokeWidth={2.5} />
      </WindowButton>
      <WindowButton title="Schließen" size={34} radius={10} danger onClick={onClose}>
        <Icon paths='<path d="M18 6 6 18"/><path d="m6 6 12 12"/>' size={14} strokeWidth={2.5} />
      </WindowButton>
    </div>
  );
}
