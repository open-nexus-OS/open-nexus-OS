import React, { useState, useRef, useEffect } from 'react';

/**
 * Menu — glass dropdown/context menu for AIVA OS (maps to ArkUI Menu / formMenu).
 * A frosted panel of MenuItems anchored to a trigger; opens on click (or on
 * right-click when contextMenu is set). Supports icons, shortcuts, submenus,
 * dividers, destructive items. Dismisses on outside-click / Escape.
 */
export function Menu({
  trigger,
  items = [],
  open,
  onOpenChange,
  placement = 'bottom-start',
  contextMenu = false,
  width = 220,
  style = {},
}) {
  const [internal, setInternal] = useState(false);
  const isOpen = open !== undefined ? open : internal;
  const [coords, setCoords] = useState(null); // {x,y} for context menu
  const wrapRef = useRef(null);

  const setOpen = (v) => { if (open === undefined) setInternal(v); onOpenChange && onOpenChange(v); };

  useEffect(() => {
    if (!isOpen) return;
    const onDoc = (e) => { if (wrapRef.current && !wrapRef.current.contains(e.target)) setOpen(false); };
    const onKey = (e) => { if (e.key === 'Escape') setOpen(false); };
    document.addEventListener('mousedown', onDoc);
    document.addEventListener('keydown', onKey);
    return () => { document.removeEventListener('mousedown', onDoc); document.removeEventListener('keydown', onKey); };
  }, [isOpen]);

  const [vert, horiz] = placement.split('-');
  const submenuLeft = horiz === 'end';
  const panelPos = contextMenu && coords
    ? { position: 'fixed', top: coords.y, left: coords.x }
    : {
        position: 'absolute',
        ...(vert === 'top' ? { bottom: 'calc(100% + 6px)' } : { top: 'calc(100% + 6px)' }),
        ...(horiz === 'end' ? { right: 0 } : { left: 0 }),
      };

  const onTriggerClick = (e) => {
    if (contextMenu) return;
    setOpen(!isOpen);
  };
  const onTriggerCtx = (e) => {
    if (!contextMenu) return;
    e.preventDefault();
    setCoords({ x: e.clientX, y: e.clientY });
    setOpen(true);
  };

  const panelStyle = {
    width, padding: 6,
    background: 'var(--glass-overlay-bg)',
    border: '1px solid var(--glass-overlay-border)',
    borderRadius: 'var(--radius-xl)',
    boxShadow: 'var(--glass-overlay-shadow)',
    backdropFilter: 'blur(40px) saturate(180%)', WebkitBackdropFilter: 'blur(40px) saturate(180%)',
    transformOrigin: (vert === 'top' ? 'bottom' : 'top') + ' ' + (horiz === 'end' ? 'right' : 'left'),
    animation: 'nx-menu-in var(--motion-quick, 0.28s) var(--motion-spring, cubic-bezier(0.34,1.4,0.5,1)) both',
  };

  const Item = ({ it }) => {
    const [hover, setHover] = useState(false);
    const [subOpen, setSubOpen] = useState(false);
    if (it.divider) return <div style={{ height: 1, background: 'var(--glass-divider)', margin: '5px 8px' }} />;
    if (it.label && it.header) return (
      <div style={{ padding: '6px 11px 4px', fontFamily: 'var(--font-sans)', fontSize: 11, fontWeight: 600, letterSpacing: '0.06em', textTransform: 'uppercase', color: 'var(--glass-text-secondary)' }}>{it.label}</div>
    );
    const hasSub = Array.isArray(it.submenu) && it.submenu.length > 0;
    return (
      <div
        style={{ position: 'relative' }}
        onMouseEnter={() => { setHover(true); if (hasSub) setSubOpen(true); }}
        onMouseLeave={() => { setHover(false); if (hasSub) setSubOpen(false); }}
      >
        <button
          onClick={() => {
            if (it.disabled) return;
            if (hasSub) { setSubOpen((o) => !o); return; }
            it.onSelect && it.onSelect();
            setOpen(false);
          }}
          disabled={it.disabled}
          style={{
            display: 'flex', alignItems: 'center', gap: 10, width: '100%',
            padding: '8px 11px', border: 'none', cursor: it.disabled ? 'not-allowed' : 'pointer', textAlign: 'left',
            borderRadius: 'var(--radius-md)', opacity: it.disabled ? 0.4 : 1,
            background: (hover && !it.disabled) ? 'var(--glass-hover-bg)' : 'transparent',
            fontFamily: 'var(--font-sans)', fontSize: 14, fontWeight: 500,
            color: it.destructive ? 'var(--color-destructive)' : 'var(--glass-text-strong)',
            transition: 'background 0.12s ease', WebkitTapHighlightColor: 'transparent',
          }}
        >
          {it.icon && <span style={{ display: 'flex', flexShrink: 0, width: 17, opacity: 0.85 }}>{it.icon}</span>}
          <span style={{ flex: 1, whiteSpace: 'nowrap', overflow: 'hidden', textOverflow: 'ellipsis' }}>{it.label}</span>
          {it.shortcut && <span style={{ fontSize: 12, color: 'var(--glass-text-secondary)', fontVariantNumeric: 'tabular-nums', flexShrink: 0 }}>{it.shortcut}</span>}
          {hasSub && (
            <svg width="14" height="14" viewBox="0 0 24 24" fill="none" style={{ flexShrink: 0, color: 'var(--glass-text-secondary)', opacity: 0.7 }}>
              <path d={submenuLeft ? 'M15 6l-6 6 6 6' : 'M9 6l6 6-6 6'} stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
            </svg>
          )}
          {!hasSub && it.checked && (
            <svg width="14" height="14" viewBox="0 0 24 24" fill="none" style={{ flexShrink: 0, color: 'var(--color-info)' }}>
              <path d="M4 12.5l5 5L20 6.5" stroke="currentColor" strokeWidth="2.6" strokeLinecap="round" strokeLinejoin="round" />
            </svg>
          )}
        </button>
        {hasSub && subOpen && (
          <div
            role="menu"
            style={{
              position: 'absolute', top: -6, zIndex: 1,
              ...(submenuLeft ? { right: '100%' } : { left: '100%' }),
              ...panelStyle,
            }}
          >
            {it.submenu.map((s, i) => <Item key={i} it={s} />)}
          </div>
        )}
      </div>
    );
  };

  return (
    <div ref={wrapRef} style={{ position: 'relative', display: 'inline-flex' }} onContextMenu={onTriggerCtx}>
      <span onClick={onTriggerClick} style={{ display: 'inline-flex' }}>{trigger}</span>
      {isOpen && (
        <div
          role="menu"
          style={{
            ...panelPos, zIndex: 'var(--z-overlay, 20)',
            ...panelStyle,
            ...style,
          }}
        >
          {items.map((it, i) => <Item key={i} it={it} />)}
        </div>
      )}
    </div>
  );
}

/**
 * ContextMenu — convenience wrapper: right-click `children` to open `items`.
 */
export function ContextMenu({ children, items, width, style }) {
  return <Menu contextMenu trigger={children} items={items} width={width} style={style} />;
}
